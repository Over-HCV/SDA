\section{Distancia y calidad de una partición}
% =============================================================================

\begin{frame}{Métodos duros y suaves}
  \centering\small
  \begin{tabular}{@{}lll@{}}
    \toprule
    & Duros & Suaves \\
    \midrule
    Numéricas    & K-medias, jerárquico & GMM \\
    Cualitativas & \textbf{K-modas}, \textbf{Gower + jerárquico/PAM}
                 & \textbf{Clases latentes} \\
    Salida       & $g_i \in \{1,\dots,K\}$ & $\boldsymbol\pi_i=(\pi_{i1},\dots,\pi_{iK})$ \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{Distancia entre categorías}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \[
        d(\mathbf{x}_i,\mathbf{x}_j) = \sum_{k=1}^{p} \mathbb{1}[x_{ik} \neq x_{jk}]
      \]
      \footnotesize El emparejamiento simple (Hamming) cuenta en cuántos
      atributos difieren dos animales. Como todas las variables ya están en
      la misma escala \textbf{no hace falta estandarizar}.
    \end{column}
    \begin{column}{0.5\textwidth}
      \[
        s_{ij} = \frac{\sum_k w_{ijk}\, s_{ijk}}{\sum_k w_{ijk}}
      \]
      \footnotesize Gower \parencite{gower1971} normaliza cada numérica por
      su rango (su forma de estandarizar) cuando hay datos mixtos y en las
      binarias asimétricas no cuenta como parecido que los dos animales
      tengan un no.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Calidad de la partición}
  La fórmula de la clase 5 solo usa distancias y no necesita un centro:
  \[
    d(\text{Agrupamiento}) = \sum_{k=1}^{K} \frac{1}{n_k}
      \sum_{i,j \in G_k} d(\mathbf{x}_i,\mathbf{x}_j)
  \]
  \begin{itemize}
    \item Con $d$ igual a la euclídea al cuadrado es el $d^2$ de la clase.
    \item Con $d$ igual a los desacuerdos funciona igual, así que
      \textbf{sirve con cualquier distancia}.
  \end{itemize}
\end{frame}

\section{K-modas}
% =============================================================================

\begin{frame}{Algoritmo k-modas}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{enumerate}
        \item Inicialización: Se toman $K$ animales como modas iniciales.
        \item Asignación: Cada animal va a la moda con la que tiene menos
          desacuerdos.
        \item Actualización: Se recalcula la moda de cada atributo dentro
          del grupo.
        \item Se repite hasta que ningún animal cambie de grupo.
      \end{enumerate}
    \end{column}
    \begin{column}{0.45\textwidth}
      \[
        \sum_{i \in G} d(\mathbf{x}_i,\mathbf{m})
          = \sum_{k}\bigl(n_G - n_k(m_k)\bigr)
      \]
      \footnotesize Es el algoritmo de Lloyd con la moda como centro. El
      costo se separa por atributo y lo minimiza la categoría más frecuente
      \parencite{huang1998}.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{K-modas en R}
  \begin{columns}[c]
    \begin{column}{0.61\textwidth}
\begin{lstlisting}
# Asignación: Cada animal a la moda con menos desacuerdos
d <- colSums(t(modas) != codigos[i, ])
destino <- which.min(d)
# Actualización: La categoría más frecuente de cada atributo
modas_de <- function(g) t(sapply(seq_len(k), function(j)
  sapply(seq_len(ncol(codigos)), function(v)
    which.max(tabulate(codigos[g == j, v], niveles[v])))))
# Costo: Desacuerdos de cada animal con la moda de su grupo
costo <- function(g, m) sum(codigos != m[g, , drop = FALSE])

# 25 arranques (semillas 42 a 66) y gana el de menor costo
kmodas7 <- kmodas(cuali, k = 7)
kmodas7$costo   # 138
\end{lstlisting}

\section{Métodos suaves: Clases latentes}
% =============================================================================

\begin{frame}{Modelo de clases latentes}
  \[
    P(\mathbf{x}_i) = \sum_{c=1}^{K} \pi_c \prod_{k=1}^{p} \prod_{r}
      \theta_{krc}^{\,\mathbb{1}[x_{ik}=r]}
  \]
  \begin{itemize}
    \item Donde el GMM pone una normal, aquí va un producto de categóricas,
      lo que supone \textbf{independencia local}.
    \item Se estima con EM, que alterna la probabilidad de pertenencia
      $\pi_{ic}$ con las nuevas $\pi_c$ y $\theta_{krc}$.
    \item Elegimos $K$ con el BIC \parencite{linzer2011}.
  \end{itemize}
\end{frame}

\begin{frame}{Número de clases: BIC y AIC}
  \centering\figura{lca-bic.png}{0.72\textheight}

  \footnotesize El BIC elige 4 clases y el AIC, que castiga menos cada
  parámetro, elige 7, que es justo el número de tipos de Zoo.
\end{frame}

\begin{frame}[fragile]{Clases latentes en R: El algoritmo EM}
  \begin{columns}[c]
    \begin{column}{0.61\textwidth}
\begin{lstlisting}
# Paso E: Posterior de cada animal en cada clase
log_f <- Y %*% log(theta) +
  rep(log(pesos), each = nrow(Y))
tope <- apply(log_f, 1, max)
suma <- rowSums(exp(log_f - tope))
posterior <- exp(log_f - tope - log(suma))
# Paso M: Pesos y probabilidades de cada clase
masa  <- colSums(posterior)
theta <- sweep(crossprod(Y, posterior), 2, masa, "/")
pesos <- masa / sum(masa)
# BIC: Cada clase de más suma 21 parámetros
bic <- -2 * ll + n_par * log(nrow(cuali))
\end{lstlisting}

      {\tiny\color{mcmBlack!60} Resumido de \texttt{em\_latente()} en el
      cuaderno; $Y$ es la tabla indicadora.}

\section{Limitaciones y conclusiones}
% =============================================================================

\begin{frame}{Limitaciones de cada método}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{K-modas}
        \footnotesize
        \begin{lista}
          \item Depende del arranque.
          \item Hay empates en la moda.
          \item Todos los atributos pesan igual.
          \item Pierde el orden de las ordinales.
        \end{lista}
      \end{alertblock}
      \begin{alertblock}{Gower}
        \footnotesize
        \begin{lista}
          \item Todos los atributos pesan igual.
          \item Hay que decidir qué binarias son asimétricas.
          \item Ocupa memoria $O(n^2)$.
        \end{lista}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Clases latentes}
        \footnotesize
        \begin{lista}
          \item Supone independencia local, que falla con pelo y leche.
          \item Puede quedar en un máximo local.
          \item Aparecen celdas con $\theta$ igual a 0 o 1.
        \end{lista}
      \end{alertblock}
      \begin{alertblock}{Tándem}
        \footnotesize Las dimensiones del ACM resumen inercia y nada
        asegura que separen los grupos.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
