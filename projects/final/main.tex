\section{Distancia y calidad de una partición}
% =============================================================================

\begin{frame}{Duros y suaves, igual que en la sesión 5}
  \centering\small
  \begin{tabular}{@{}lll@{}}
    \toprule
    & Duros & Suaves \\
    \midrule
    Numéricas    & K-medias, jerárquico & GMM \\
    Cualitativas & \textbf{K-modas}, \textbf{Gower + jerárquico/PAM}
                 & \textbf{Clases latentes} \\
    Salida       & $g_i \in \{1,\dots,K\}$ & $\boldsymbol\pi_i=(\pi_{i1},\dots,\pi_{iK})$ \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{De la euclídea a los desacuerdos}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \[
        d(\mathbf{x}_i,\mathbf{x}_j) = \sum_{k=1}^{p} \mathbb{1}[x_{ik} \neq x_{jk}]
      \]
      \footnotesize Emparejamiento simple (Hamming): en cuántos atributos
      difieren. Todas las variables ya están en la misma escala:
      \textbf{no hay nada que estandarizar}.
    \end{column}
    \begin{column}{0.5\textwidth}
      \[
        s_{ij} = \frac{\sum_k w_{ijk}\, s_{ijk}}{\sum_k w_{ijk}}
      \]
      \footnotesize Gower \parencite{gower1971}: con datos mixtos normaliza
      cada numérica por su rango (su «estandarizar»); con binarias
      asimétricas dos «no» no cuentan.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Calidad de la partición}
  La fórmula de la sesión 5 no necesita un centro, solo distancias:
  \[
    d(\text{Agrupamiento}) = \sum_{k=1}^{K} \frac{1}{n_k}
      \sum_{i,j \in G_k} d(\mathbf{x}_i,\mathbf{x}_j)
  \]
  \begin{itemize}
    \item Con $d$ = euclídea al cuadrado es el $d^2$ de clase.
    \item Con $d$ = desacuerdos funciona igual: \textbf{sirve con cualquier
      distancia}.
  \end{itemize}
\end{frame}

\section{K-modas}
% =============================================================================

\begin{frame}{Lloyd con otro centro}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{enumerate}
        \item Inicialización: $K$ animales como modas iniciales.
        \item Asignación: cada animal a la moda con menos desacuerdos.
        \item Actualización: la moda de cada atributo en el grupo.
        \item Repetir hasta que nadie cambie.
      \end{enumerate}
    \end{column}
    \begin{column}{0.45\textwidth}
      \[
        \sum_{i \in G} d(\mathbf{x}_i,\mathbf{m})
          = \sum_{k}\bigl(n_G - n_k(m_k)\bigr)
      \]
      \footnotesize El costo se separa por atributo, y lo minimiza la
      categoría más frecuente \parencite{huang1998}.
    \end{column}
  \end{columns}
\end{frame}

\section{Métodos suaves: clases latentes}
% =============================================================================

\begin{frame}{El GMM de las categorías}
  \[
    P(\mathbf{x}_i) = \sum_{c=1}^{K} \pi_c \prod_{k=1}^{p} \prod_{r}
      \theta_{krc}^{\,\mathbb{1}[x_{ik}=r]}
  \]
  \begin{itemize}
    \item Donde el GMM pone una normal, aquí va un producto de categóricas
      (\textbf{independencia local}).
    \item EM: probabilidad de pertenencia $\pi_{ic}$ → nuevas $\pi_c$ y
      $\theta_{krc}$.
    \item $K$ por BIC \parencite{linzer2011}.
  \end{itemize}
\end{frame}

\begin{frame}{¿Cuántas clases?}
  \centering\figura{lca-bic.png}{0.72\textheight}

  \footnotesize El BIC elige 4 clases; el AIC, que castiga menos los
  parámetros, elige 7, justo los tipos de Zoo.
\end{frame}

% =============================================================================
\section{Supuestos, fallos y síntesis}
% =============================================================================

\begin{frame}{Dónde se rompe cada uno}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{K-modas}
        \footnotesize Depende del arranque · empates de moda · pesos iguales ·
        pierde el orden de las ordinales.
      \end{alertblock}
      \begin{alertblock}{Gower}
        \footnotesize Pesos iguales · binarias asimétricas · memoria $O(n^2)$.
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Clases latentes}
        \footnotesize Independencia local (pelo y leche) · máximos locales ·
        celdas con $\theta = 0$ o $1$.
      \end{alertblock}
      \begin{alertblock}{Tándem}
        \footnotesize Las dimensiones del ACM no buscan separar grupos.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Para llevarse}
  \begin{itemize}
    \item El arco de la sesión 5 se mantiene; cambian la distancia, el centro
      o el modelo.
    \item Duros (k-modas, Gower) y suaves (clases latentes) también existen
      para categorías.
    \item El ACM y el clustering son \textbf{complementarios, no
      sustitutos}.
  \end{itemize}
  \begin{exampleblock}{Una frase}
    Con datos cualitativos no se agrupa distinto: se mide distinto.
  \end{exampleblock}
\end{frame}
